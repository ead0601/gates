# Basic variable workflow
set names1 = U1.A0 "U1.A 1" U2.B
set names2 = A.B C.D
vars
vars names1
echo $names1 | awk '{print NF " items"}'
unset names2
vars

# Inventory (use supported subcommands)
list files
list components --limit 10
list instances --limit 10
list pins U1 --like 'U1.*'
list port --like 'OUT*'
list nets --limit 10

# Find (category + pattern)
find instance U1*
find port OUT*
find net U*
find components REG*

# Paths (use required flags)
paths --from U1.A0 --to U2.B --depth 50 --max-paths 10

# Fanin / Fanout
fanin U1.A0 --tree
fanout U1.A0 --tree

# Show a couple of things
show U1
show U2
